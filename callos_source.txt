        document.addEventListener('DOMContentLoaded', function () {
            function initRichLandingCarousel(config) {
                var carousel = config.carousel;
                var trackSelector = config.trackSelector;
                var getVisibleCount = config.getVisibleCount;
                var shell = config.shell || carousel.parentElement;
                var pagination = shell ? shell.querySelector(config.paginationSelector) : null;
                var prevBtn = shell ? shell.querySelector(config.prevSelector) : null;
                var nextBtn = shell ? shell.querySelector(config.nextSelector) : null;
                var dotClass = config.dotClass || 'landing-carousel-dot';
                var onLayout = config.onLayout;

                var controls = initLandingCarousel(carousel, trackSelector, getVisibleCount, onLayout);
                if (!controls) return null;

                var track = controls.getTrack();
                var cards = Array.from(track.children);
                cards.forEach(function (card, index) {
                    card.setAttribute('data-carousel-index', String(index));
                });

                var totalSlides = cards.length;
                var showCount = typeof getVisibleCount === 'function' ? getVisibleCount() : 3;
                if (totalSlides <= showCount) {
                    if (pagination) pagination.style.display = 'none';
                    if (prevBtn) prevBtn.style.display = 'none';
                    if (nextBtn) nextBtn.style.display = 'none';
                    return controls;
                }

                function getActiveIndex() {
                    var firstCard = track.children[0];
                    if (!firstCard) return 0;
                    var parsed = Number.parseInt(firstCard.getAttribute('data-carousel-index') || '0', 10);
                    return Number.isNaN(parsed) ? 0 : parsed;
                }

                function syncDots() {
                    if (!pagination) return;
                    var activeIndex = getActiveIndex();
                    Array.from(pagination.children).forEach(function (dot, index) {
                        var isActive = index === activeIndex;
                        dot.classList.toggle('is-active', isActive);
                        dot.setAttribute('aria-current', isActive ? 'true' : 'false');
                    });
                }

                function jumpTo(index) {
                    var guard = 0;
                    while (getActiveIndex() !== index && guard < totalSlides) {
                        var first = track.children[0];
                        if (!first) break;
                        track.appendChild(first);
                        guard += 1;
                    }
                    track.style.transition = 'none';
                    track.style.transform = 'translateX(0)';
                    void track.offsetHeight;
                    track.style.transition = 'transform 0.55s ease';
                    syncDots();
                    controls.restart();
                }

                if (pagination) {
                    pagination.innerHTML = '';
                    cards.forEach(function (_, index) {
                        var dot = document.createElement('button');
                        dot.type = 'button';
                        dot.className = dotClass;
                        dot.setAttribute('aria-label', 'Go to testimonial ' + (index + 1));
                        dot.addEventListener('click', function () {
                            jumpTo(index);
                        });
                        pagination.appendChild(dot);
                    });
                }

                if (prevBtn) {
                    prevBtn.addEventListener('click', function () {
                        controls.prev();
                        window.setTimeout(syncDots, 560);
                    });
                }
                if (nextBtn) {
                    nextBtn.addEventListener('click', function () {
                        controls.next();
                        window.setTimeout(syncDots, 560);
                    });
                }

                track.addEventListener('transitionend', syncDots);
                syncDots();

                if (shell) {
                    shell.addEventListener('mouseenter', controls.pause);
                    shell.addEventListener('mouseleave', controls.restart);
                    shell.addEventListener('focusin', controls.pause);
                    shell.addEventListener('focusout', function (event) {
                        if (!shell.contains(event.relatedTarget)) {
                            controls.restart();
                        }
                    });
                }

                return controls;
            }

            // Same horizontal carousel as "Real Clients. Real Results" (testimonials)
            function initLandingCarousel(carousel, trackSelector, getVisibleCount, onLayout) {
                var track = carousel.querySelector(trackSelector);
                if (!track) return;

                var timerId = null;
                var animating = false;

                function visibleCount() {
                    if (typeof getVisibleCount === 'function') {
                        return getVisibleCount();
                    }
                    return window.matchMedia('(max-width: 767px)').matches ? 1 : 3;
                }

                function cardStepWidth() {
                    var first = track.children[0];
                    var second = track.children[1];
                    if (!first) return 0;
                    if (!second) return first.getBoundingClientRect().width;
                    return second.getBoundingClientRect().left - first.getBoundingClientRect().left;
                }

                function setCardWidth() {
                    var cards = Array.from(track.children);
                    var show = visibleCount();
                    var gap = 32; // 2rem — must match .callos-videos-track / .testimonials-grid gap
                    var width = carousel.clientWidth;
                    var cardWidth = (width - (gap * (show - 1))) / show;
                    cards.forEach(function (card) {
                        card.style.width = cardWidth + 'px';
                    });
                    if (typeof onLayout === 'function') {
                        onLayout();
                    } else if (trackSelector === '[data-testimonial-track]' && typeof window.refreshTestimonialReadMore === 'function') {
                        window.refreshTestimonialReadMore();
                    }
                }

                function nextSlide() {
                    if (animating || track.children.length <= visibleCount()) return;
                    var first = track.children[0];
                    var step = cardStepWidth();
                    if (!first || step <= 0) return;

                    animating = true;
                    track.style.transform = 'translateX(-' + step + 'px)';

                    var onEnd = function () {
                        track.removeEventListener('transitionend', onEnd);
                        track.style.transition = 'none';
                        track.appendChild(first);
                        track.style.transform = 'translateX(0)';
                        void track.offsetHeight;
                        track.style.transition = 'transform 0.55s ease';
                        animating = false;
                    };
                    track.addEventListener('transitionend', onEnd, { once: true });
                }

                function prevSlide() {
                    if (animating || track.children.length <= visibleCount()) return;
                    var last = track.children[track.children.length - 1];
                    var step = cardStepWidth();
                    if (!last || step <= 0) return;

                    animating = true;
                    track.style.transition = 'none';
                    track.insertBefore(last, track.children[0]);
                    track.style.transform = 'translateX(-' + step + 'px)';
                    void track.offsetHeight;
                    track.style.transition = 'transform 0.55s ease';
                    track.style.transform = 'translateX(0)';

                    var onEnd = function () {
                        track.removeEventListener('transitionend', onEnd);
                        animating = false;
                    };
                    track.addEventListener('transitionend', onEnd, { once: true });
                }

                function start() {
                    stop();
                    setCardWidth();
                    if (track.children.length <= visibleCount()) return;
                    timerId = window.setInterval(nextSlide, 6000);
                }

                function stop() {
                    if (timerId) {
                        window.clearInterval(timerId);
                        timerId = null;
                    }
                }

                track.style.transition = 'transform 0.55s ease';
                start();
                window.addEventListener('resize', function () {
                    setCardWidth();
                    start();
                });

                return {
                    next: function () {
                        nextSlide();
                        start();
                    },
                    prev: function () {
                        prevSlide();
                        start();
                    },
                    restart: start,
                    pause: stop,
                    getTrack: function () {
                        return track;
                    }
                };
            }

            function initTestimonialReadMore() {
                var modal = document.querySelector('[data-testimonial-modal]');
                if (!modal) return;

                var modalMessage = modal.querySelector('[data-testimonial-modal-message]');
                var modalName = modal.querySelector('[data-testimonial-modal-name]');
                var modalRole = modal.querySelector('[data-testimonial-modal-role]');
                var modalAvatar = modal.querySelector('[data-testimonial-modal-avatar]');
                var modalBadge = modal.querySelector('[data-testimonial-modal-badge]');
                var closeBtn = modal.querySelector('[data-testimonial-modal-close]');

                function getTestimonialLineHeight(textEl) {
                    var style = window.getComputedStyle(textEl);
                    var lineHeight = parseFloat(style.lineHeight);
                    if (!isNaN(lineHeight)) return lineHeight;
                    return parseFloat(style.fontSize) * 1.6;
                }

                function getNaturalTextHeight(textEl) {
                    var width = textEl.getBoundingClientRect().width;
                    if (width <= 0) return 0;

                    var clone = textEl.cloneNode(true);
                    clone.classList.remove('is-clamped');
                    clone.style.cssText = [
                        'position:absolute',
                        'visibility:hidden',
                        'pointer-events:none',
                        'height:auto',
                        'max-height:none',
                        'overflow:visible',
                        'display:block',
                        'width:' + width + 'px'
                    ].join(';');
                    textEl.parentNode.appendChild(clone);
                    var naturalHeight = clone.offsetHeight;
                    clone.remove();
                    return naturalHeight;
                }

                function getAvailableTextHeight(card, textEl, reserveReadMore) {
                    var wrap = card.querySelector('.testimonial-text-wrap');
                    var footer = card.querySelector('.testimonial-card-footer');
                    if (!wrap || !footer) return 0;

                    var available = footer.getBoundingClientRect().top - wrap.getBoundingClientRect().top;
                    if (reserveReadMore) {
                        var moreWrap = card.querySelector('.testimonial-text-more');
                        var wasHidden = moreWrap && moreWrap.hidden;
                        if (moreWrap && wasHidden) {
                            moreWrap.hidden = false;
                        }
                        var moreHeight = moreWrap ? moreWrap.offsetHeight : 28;
                        if (moreWrap && wasHidden) {
                            moreWrap.hidden = true;
                        }
                        available -= moreHeight + 6;
                    }
                    return Math.max(0, available);
                }

                function getLineAlignedMaxHeight(textEl, availableHeight) {
                    var lineH = getTestimonialLineHeight(textEl);
                    if (lineH <= 0) return availableHeight;
                    var lines = Math.max(1, Math.floor(availableHeight / lineH));
                    return lines * lineH;
                }

                function refreshReadMoreButtons() {
                    document.querySelectorAll('[data-featured-testimonial]').forEach(function (card) {
                        var textEl = card.querySelector('.featured-testimonial-message');
                        var moreWrap = card.querySelector('.featured-testimonial-text-more');
                        if (!textEl || !moreWrap) return;

                        var lineHeight = getTestimonialLineHeight(textEl);
                        var clampHeight = lineHeight * 5;
                        moreWrap.hidden = textEl.scrollHeight <= clampHeight + 2;
                    });

                    document.querySelectorAll('[data-testimonial-card]:not([data-featured-testimonial])').forEach(function (card) {
                        var textEl = card.querySelector('.testimonial-text');
                        var moreWrap = card.querySelector('.testimonial-text-more');
                        if (!textEl || !moreWrap) return;

                        textEl.classList.remove('is-clamped');
                        textEl.style.maxHeight = '';

                        var naturalHeight = getNaturalTextHeight(textEl);
                        var fullAvail = getAvailableTextHeight(card, textEl, false);
                        var truncated = naturalHeight > getLineAlignedMaxHeight(textEl, fullAvail) + 1;

                        if (truncated) {
                            var clampHeight = getLineAlignedMaxHeight(textEl, getAvailableTextHeight(card, textEl, true));
                            textEl.classList.add('is-clamped');
                            textEl.style.maxHeight = clampHeight + 'px';
                            moreWrap.hidden = false;
                        } else {
                            textEl.classList.remove('is-clamped');
                            textEl.style.maxHeight = '';
                            moreWrap.hidden = true;
                        }
                    });
                }

                window.refreshTestimonialReadMore = refreshReadMoreButtons;

                function openTestimonialModal(card) {
                    if (!card) return;
                    var textEl = card.querySelector('.testimonial-text');
                    var nameEl = card.querySelector('.testimonial-name');

                    }
                    if (modalName) {
                        modalName.textContent = nameEl ? nameEl.textContent.trim() : '';
                    }
                    if (modalRole) {
                        modalRole.textContent = roleEl ? roleEl.textContent.trim() : '';
                    }
                    if (modalAvatar && avatarEl) {
                        modalAvatar.innerHTML = avatarEl.innerHTML;
                    }
                    if (modalBadge) {
                        var industry = badgeEl ? badgeEl.textContent.trim() : '';
                        if (industry) {
                            modalBadge.textContent = industry;
                            modalBadge.hidden = false;
                        } else {
                            modalBadge.textContent = '';
                            modalBadge.hidden = true;
                        }
                    }

                    modal.classList.add('is-open');
                    modal.setAttribute('aria-hidden', 'false');
                    document.body.style.overflow = 'hidden';
                }

                function closeTestimonialModal() {
                    modal.classList.remove('is-open');
                    modal.setAttribute('aria-hidden', 'true');
                    document.body.style.overflow = '';
                }

                document.querySelectorAll('[data-testimonial-card] .testimonial-read-more').forEach(function (btn) {
                    btn.addEventListener('click', function (e) {
                        e.preventDefault();
                        e.stopPropagation();
                        openTestimonialModal(btn.closest('[data-testimonial-card]'));
                    });
                });

                document.fonts && document.fonts.ready.then(refreshReadMoreButtons);

                if (closeBtn) {
                    closeBtn.addEventListener('click', closeTestimonialModal);
                }

                modal.addEventListener('click', function (e) {
                    if (e.target === modal) {
                        closeTestimonialModal();
                    }
                });

                document.addEventListener('keydown', function (e) {
                    if (e.key === 'Escape' && modal.classList.contains('is-open')) {
                        closeTestimonialModal();
                    }
                });

                refreshReadMoreButtons();
                requestAnimationFrame(refreshReadMoreButtons);
                window.setTimeout(refreshReadMoreButtons, 350);
                window.addEventListener('resize', refreshReadMoreButtons);
            }

            document.querySelectorAll('[data-testimonial-carousel]').forEach(function (carousel) {
                initRichLandingCarousel({
